# 点云与测距

> 用 D435i 做点云生成、物体测距、体积估算的实战方法。

## 1. 点云基础

### 1.1 三种坐标系

D435i 生成点云时涉及三个坐标系：

```
camera_depth_frame  →  深度图像坐标系
        │
        │ 内参 fx, fy, cx, cy（深度内参）
        ↓
camera_depth_optical_frame（OpenCV 风格）
        │
        │ 外参（深度 → RGB / IMU）
        ↓
camera_color_frame / camera_imu_frame
```

### 1.2 点云生成公式

对深度图中像素 (u, v)，其 3D 坐标为：

```
X = (u - cx) * Z / fx
Y = (v - cy) * Z / fy
Z = depth_image[v, u] * depth_units (默认 0.001 米)
```

### 1.3 点云质量

| 因素 | 影响 |
|------|------|
| 深度分辨率 | 1280×720 比 640×480 多 4 倍点 |
| 滤波 | 时序+空间滤波减少噪声点 |
| 距离 | 远距离点云稀疏，误差大 |
| 视场角 | 边缘视场畸变，需去畸变 |

### 2.2 用 PCL 处理点云

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import PointCloud2
import pcl
import pcl_helper  # 自定义

class CloudProcess(Node):
    def __init__(self):
        super().__init__('cloud_proc')
        self.sub = self.create_subscription(
            PointCloud2, '/camera/depth/color/points',
            self.callback, 10
        )
    
    def callback(self, msg):
        # ROS → PCL
        cloud = pcl_helper.ros_to_pcl(msg)
        # ... 处理 ...
        
        # 下采样（VoxelGrid）
        vox = cloud.make_voxel_grid_filter()
        vox.set_leaf_size(0.01, 0.01, 0.01)
        cloud_filtered = vox.filter()
        
        # 平面分割（找地面）
        seg = cloud_filtered.make_segmenter()
        seg.set_model_type(pcl.SACMODEL_PLANE)
        seg.set_method_type(pcl.SAC_RANSAC)
        seg.set_distance_threshold(0.01)
        indices, coefficients = seg.segment()
        
        # 移除地面
        cloud_no_ground = cloud_filtered.extract(indices, negative=True)
        
        # 欧氏聚类
        ec = cloud_no_ground.make_Euclidean_clusterer()
        ec.set_ClusterTolerance(0.02)
        ec.set_MinClusterSize(100)
        ec.set_MaxClusterSize(25000)
        cluster_indices = ec.extract()
        
        self.get_logger().info(f'检测到 {len(cluster_indices)} 个聚类')
```

## 4. 测距：像素 → 米

### 4.1 单点测距

```python
import pyrealsense2 as rs

# 获取内参
profile = pipe.get_active_profile()
depth_profile = profile.get_stream(rs.stream.depth)
intrinsics = depth_profile.as_video_stream_profile().get_intrinsics()

# 像素 → 3D
def pixel_to_3d(depth_frame, x, y):
    dist = depth_frame.get_distance(x, y)  # 米
    if dist == 0:
        return None  # 无效像素
    
    pt_3d = rs.rs2_deproject_pixel_to_point(intrinsics, [x, y], dist)
    return pt_3d  # [X, Y, Z] 米

# 示例：测图像中心
pt = pixel_to_3d(depth_frame, 320, 240)
print(f"中心点 3D: {pt}")
```

### 4.2 区域测距（最近/平均/最远）

```python
import numpy as np

def region_depth_stats(depth_image, x1, y1, x2, y2):
    """
    给定矩形区域，返回深度统计
    depth_image: numpy (H, W) uint16
    """
    roi = depth_image[y1:y2, x1:x2]
    valid = roi[roi > 0]  # 排除 0 = 无效
    
    if len(valid) == 0:
        return None
    
    return {
        'min': float(valid.min()) / 1000,   # 米
        'max': float(valid.max()) / 1000,
        'mean': float(valid.mean()) / 1000,
        'std': float(valid.std()) / 1000,
        'count': len(valid),
        'total': roi.size
    }

# 示例：前方 100×100 像素区域
stats = region_depth_stats(depth_image, 270, 190, 370, 290)
print(f"前方区域: min={stats['min']:.2f} mean={stats['mean']:.2f}")
```

### 4.3 ROS2 服务：实时测距

```python
from rclpy.node import Node
from sensor_msgs.msg import Image
from your_msgs.srv import MeasureDistance, MeasureDistanceResponse

class MeasureService(Node):
    def __init__(self):
        super().__init__('measure_distance')
        
        self.depth_sub = self.create_subscription(
            Image, '/camera/depth/image_rect_raw',
            self.depth_callback, 10
        )
        self.latest_depth = None
        
        self.srv = self.create_service(
            MeasureDistance, '~/measure_at_pixel',
            self.measure_callback
        )
    
    def measure_callback(self, request, response):
        # request.pixel_x, pixel_y
        if self.latest_depth is None:
            response.distance = -1.0
            return response
        
        h, w = self.latest_depth.shape
        x = int(request.pixel_x)
        y = int(request.pixel_y)
        
        if 0 <= x < w and 0 <= y < h:
            dist_mm = self.latest_depth[y, x]
            response.distance = dist_mm / 1000.0 if dist_mm > 0 else -1.0
        else:
            response.distance = -1.0
        
        return response
```

## 5. 体积估算

### 5.1 简单方案：包围盒

```python
import open3d as o3d
import numpy as np

def estimate_volume_bbox(pcd):
    """包围盒体积估算"""
    bbox = pcd.get_axis_aligned_bounding_box()
    extents = bbox.get_extent()  # [x, y, z]
    volume = extents[0] * extents[1] * extents[2]
    return volume

# 更紧的 OBB（定向包围盒）
def estimate_volume_obb(pcd):
    obb = pcd.get_oriented_bounding_box()
    extents = obb.extent
    return extents[0] * extents[1] * extents[2]
```

### 5.2 进阶：凸包

```python
def estimate_volume_convex(pcd):
    """凸包体积（更紧）"""
    hull, _ = pcd.compute_convex_hull()
    volume = hull.get_volume()
    return volume
```

### 5.3 实际用例

```python
# 1. 检测桌面上的物体
pc = rs.pointcloud()
depth = frames.get_depth_frame()
points = pc.calculate(depth)

vtx = np.asanyarray(points.get_vertices())
pcd = o3d.geometry.PointCloud()
pcd.points = o3d.utility.Vector3dVector(vtx[:, :3])

# 2. 移除离群点
pcd = pcd.voxel_down_sample(0.005)
cl, ind = pcd.remove_statistical_outlier(nb_neighbors=20, std_ratio=2.0)
pcd = pcd.select_by_index(ind)

# 3. 移除地面
plane_model, inliers = pcd.segment_plane(
    distance_threshold=0.01, ransac_n=3, num_iterations=1000
)
pcd_no_ground = pcd.select_by_index(inliers, invert=True)

# 4. 聚类
labels = np.array(pcd_no_ground.cluster_dbscan(eps=0.02, min_points=10))
max_label = labels.max()

for i in range(max_label + 1):
    cluster = pcd_no_ground.select_by_index(np.where(labels == i)[0])
    vol = estimate_volume_convex(cluster)
    print(f"聚类 {i}: {vol * 1e6:.0f} cm³, {len(cluster.points)} 个点")
```

## 6. 距离精度与误差

### 6.1 误差来源

| 来源 | 量级 | 解决方案 |
|------|------|---------|
| 量化（Z16 1mm）| 0.05-5 mm | 不可消除 |
| 立体匹配 | 1-3% 距离 | 滤波 |
| 镜头畸变 | 边缘 ~2% | 去畸变 |
| 投射器失效 | 远距离大 | 对准 |
| 温度漂移 | 0.1%/℃ | 等待预热 |

### 6.2 实测精度

| 距离 | 典型误差 |
|------|---------|
| 0.3 m | ±2 mm |
| 1.0 m | ±10 mm |
| 2.0 m | ±30 mm |
| 3.0 m | ±80 mm |
