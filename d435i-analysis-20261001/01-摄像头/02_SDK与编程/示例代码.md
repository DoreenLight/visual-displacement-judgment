## 1. 示例 1：基础流采集

```python
"""
example_01_basic.py
启动深度 + 彩色流，显示实时画面
"""
import pyrealsense2 as rs
import numpy as np
import cv2

pipeline = rs.pipeline()
config = rs.config()

config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)

pipeline.start(config)

try:
    while True:
        frames = pipeline.wait_for_frames()
        depth_frame = frames.get_depth_frame()
        color_frame = frames.get_color_frame()
        
        if not depth_frame or not color_frame:
            continue
        
        # 转为 numpy
        depth_image = np.asanyarray(depth_frame.get_data())
        color_image = np.asanyarray(color_frame.get_data())
        
        # 深度图上色（便于显示）
        depth_colormap = cv2.applyColorMap(
            cv2.convertScaleAbs(depth_image, alpha=0.03), cv2.COLORMAP_JET
        )
        
        # 拼接显示
        images = np.hstack((color_image, depth_colormap))
        cv2.namedWindow('RealSense', cv2.WINDOW_AUTOSIZE)
        cv2.imshow('RealSense', images)
        
        key = cv2.waitKey(1)
        if key & 0xFF == ord('q'):
            break
finally:
    pipeline.stop()
    cv2.destroyAllWindows()
```

## 2. 示例 2：点击图像测距

```python
"""
example_02_click_distance.py
点击彩色图像，显示该点深度（米）
"""
import pyrealsense2 as rs
import numpy as np
import cv2

pipeline = rs.pipeline()
config = rs.config()
config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
pipeline.start(config)

# 对齐：让深度与彩色对齐
align = rs.align(rs.stream.color)

# 鼠标点击回调
def mouse_callback(event, x, y, flags, param):
    if event == cv2.EVENT_LBUTTONDOWN:
        depth = param['depth_frame']
        intrinsics = param['intrinsics']
        dist = depth.get_distance(x, y)
        pt_3d = rs.rs2_deproject_pixel_to_point(intrinsics, [x, y], dist)
        print(f"像素 ({x}, {y}) → 距离 {dist:.3f} m → 3D [{pt_3d[0]:.3f}, {pt_3d[1]:.3f}, {pt_3d[2]:.3f}]")

cv2.namedWindow('Color')
cv2.setMouseCallback('Color', mouse_callback)

try:
    while True:
        frames = pipeline.wait_for_frames()
        aligned = align.process(frames)
        
        depth_frame = aligned.get_depth_frame()
        color_frame = aligned.get_color_frame()
        
        color_image = np.asanyarray(color_frame.get_data())
        intrinsics = depth_frame.profile.as_video_stream_profile().intrinsics
        
        cv2.imshow('Color', color_image)
        
        # 把帧上下文传给回调
        cv2.setMouseCallback('Color', mouse_callback, 
                            {'depth_frame': depth_frame, 'intrinsics': intrinsics})
        
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
finally:
    pipeline.stop()
    cv2.destroyAllWindows()
```

## 3. 示例 3：点云生成与可视化

```python
"""
example_03_pointcloud.py
生成点云，用 Open3D 可视化
"""
import pyrealsense2 as rs
import numpy as np
import open3d as o3d

pipeline = rs.pipeline()
config = rs.config()
config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
pipeline.start(config)

pc = rs.pointcloud()
align = rs.align(rs.stream.color)

vis = o3d.visualization.Visualizer()
vis.create_window('PointCloud')

try:
    while True:
        frames = pipeline.wait_for_frames()
        aligned = align.process(frames)
        
        depth_frame = aligned.get_depth_frame()
        color_frame = aligned.get_color_frame()
        
        # 生成点云
        points = pc.calculate(depth_frame)
        # 附加颜色
        pc.map_to(color_frame)
        
        vtx = np.asanyarray(points.get_vertices())
        tex = np.asanyarray(points.get_texture_coordinates())
        
        # 用纹理坐标取色
        color_img = np.asanyarray(color_frame.get_data())
        h, w, _ = color_img.shape
        
        # Open3D 颜色格式 RGB
        colors = np.zeros((len(vtx), 3), dtype=np.float32)
        for i in range(len(vtx)):
            u = int(tex[i][0] * w)
            v = int(tex[i][1] * h)
            if 0 <= u < w and 0 <= v < h:
                colors[i] = color_img[v, u, ::-1] / 255.0
        
        # 构造 Open3D 点云
        pcd = o3d.geometry.PointCloud()
        pcd.points = o3d.utility.Vector3dVector(vtx[:, :3])
        pcd.colors = o3d.utility.Vector3dVector(colors)
        
        # 更新可视化
        vis.clear_geometries()
        vis.add_geometry(pcd)
        vis.poll_events()
        vis.update_renderer()
        
        if not vis.poll_events():
            break
finally:
    pipeline.stop()
    vis.destroy_window()
```

## 4. 示例 4：IMU 数据记录

```python
"""
example_04_imu.py
记录加速度 + 陀螺仪数据，存为 CSV
"""
import pyrealsense2 as rs
import csv
from datetime import datetime

pipeline = rs.pipeline()
config = rs.config()

# IMU 单独启用
config.enable_stream(rs.stream.accel, rs.format.motion_xyz32f, 200)
config.enable_stream(rs.stream.gyro, rs.format.motion_xyz32f, 200)
pipeline.start(config)

csv_file = open(f'imu_log_{datetime.now():%Y%m%d_%H%M%S}.csv', 'w', newline='')
writer = csv.writer(csv_file)
writer.writerow(['timestamp_ms', 'accel_x', 'accel_y', 'accel_z', 
                 'gyro_x', 'gyro_y', 'gyro_z'])

last_print = 0
try:
    while True:
        frames = pipeline.wait_for_frames()
        
        accel_frame = frames.first_or_default(rs.stream.accel)
        gyro_frame = frames.first_or_default(rs.stream.gyro)
        
        if accel_frame and gyro_frame:
            a = accel_frame.as_motion_frame().get_motion_data()
            g = gyro_frame.as_motion_frame().get_motion_data()
            t = accel_frame.get_timestamp()
            
            writer.writerow([t, a.x, a.y, a.z, g.x, g.y, g.z])
            
            # 控制台每秒输出一次
            if t - last_print > 1000:
                import math
                norm_a = math.sqrt(a.x**2 + a.y**2 + a.z**2)
                print(f"t={t:.0f}ms  |a|={norm_a:.3f} (应≈9.8)")
                last_print = t

except KeyboardInterrupt:
    print("停止")
finally:
    pipeline.stop()
    csv_file.close()
```

    while True:
        frames = pipeline.wait_for_frames()
        
        # 记录 IMU
        for stream_type in [rs.stream.accel, rs.stream.gyro]:
            f = frames.first_or_default(stream_type)
            if f:
                # 注意：实际每帧的 IMU 可能多个 sample
                pass
        
        # 记录 RGB + Depth（每 10 帧存一次）
        aligned = align.process(frames)
        depth_frame = aligned.get_depth_frame()
        color_frame = aligned.get_color_frame()

    imu_log.close()

    cv2.destroyAllWindows()

print(f"数据集已保存到 {output_dir}/")

## 6. 示例 6：检测最近障碍物距离

```python
"""
example_06_obstacle_distance.py
实时显示前方 0.5m × 0.5m 区域内的最近障碍物距离
"""
import pyrealsense2 as rs
import numpy as np
import cv2

pipeline = rs.pipeline()
config = rs.config()
config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
pipeline.start(config)

try:
    while True:
        frames = pipeline.wait_for_frames()
        depth_frame = frames.get_depth_frame()
        depth_image = np.asanyarray(depth_frame.get_data())
        
        # 取中央 1/3 区域（前方）
        h, w = depth_image.shape
        cx, cy = w // 2, h // 2
        roi = depth_image[cy-80:cy+80, cx-100:cx+100]
        
        # 找最近点（排除 0 = 无效）
        valid = roi[roi > 0]
        if len(valid) > 0:
            min_dist_mm = valid.min()
            min_dist_m = min_dist_mm / 1000.0
            print(f"前方最近障碍物: {min_dist_m:.2f} m", end='\r')
        
        # 可视化
        depth_colormap = cv2.applyColorMap(
            cv2.convertScaleAbs(depth_image, alpha=0.03), cv2.COLORMAP_JET
        )
        # 画 ROI 框
        cv2.rectangle(depth_colormap, (cx-100, cy-80), (cx+100, cy+80), (0, 255, 0), 2)
        cv2.imshow('Depth ROI', depth_colormap)
        
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
finally:
    pipeline.stop()
    cv2.destroyAllWindows()
```

## 7. 示例 7：深度图滤波

```python
"""
example_07_filter.py
应用 librealsense 滤波 pipeline
"""
import pyrealsense2 as rs
import numpy as np
import cv2

pipeline = rs.pipeline()
config = rs.config()
config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
pipeline.start(config)

# 滤波 pipeline
decimation = rs.decimation_filter(magnitude=2)  # 2x 下采样
spatial = rs.spatial_filter()
temporal = rs.temporal_filter()
hole = rs.hole_filling_filter()

colorizer = rs.colorizer()

try:
    while True:
        frames = pipeline.wait_for_frames()
        depth_frame = frames.get_depth_frame()
        
        # 滤波链
        filtered = decimation.process(depth_frame)
        filtered = spatial.process(filtered)
        filtered = temporal.process(filtered)
        filtered = hole.process(filtered)
        
        # 原始 vs 滤波后
        raw_colorized = np.asanyarray(colorizer.process(depth_frame).get_data())
        filt_colorized = np.asanyarray(colorizer.process(filtered).get_data())
        
        cv2.imshow('Raw Depth', raw_colorized)
        cv2.imshow('Filtered Depth', filt_colorized)
        
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
finally:
    pipeline.stop()
    cv2.destroyAllWindows()
```

## 8. 示例 8：ROS2 桥接（pyrealsense2 → ROS topic）

```python
"""
example_08_ros_bridge.py
用 Python pyrealsense2 + rclpy 发布 ROS2 话题
适合：不想用 realsense2_camera 包，自定义处理流程
"""
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, Imu
from cv_bridge import CvBridge
import pyrealsense2 as rs
import numpy as np
import cv2

class RealSenseBridge(Node):
    def __init__(self):
        super().__init__('realsense_bridge')
        self.pub_depth = self.create_publisher(Image, '/custom/depth', 10)
        self.pub_color = self.create_publisher(Image, '/custom/color', 10)
        self.pub_imu = self.create_publisher(Imu, '/custom/imu', 200)
        
        self.pipeline = rs.pipeline()
        config = rs.config()
        config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
        config.enable_stream(rs.stream.color, 640, 480, rs.format.bgr8, 30)
        config.enable_stream(rs.stream.accel, rs.format.motion_xyz32f, 200)
        config.enable_stream(rs.stream.gyro, rs.format.motion_xyz32f, 200)
        self.pipeline.start(config)
        
        self.bridge = CvBridge()
        self.timer = self.create_timer(0.01, self.timer_callback)  # 100Hz
    
    def timer_callback(self):
        try:
            frames = self.pipeline.wait_for_frames(timeout_ms=100)
        except RuntimeError:
            return
        
        # 深度
        depth_frame = frames.get_depth_frame()
        if depth_frame:
            depth_img = np.asanyarray(depth_frame.get_data())
            msg = self.bridge.cv2_to_imgmsg(depth_img, encoding='16UC1')
            msg.header.stamp = self.get_clock().now().to_msg()
            msg.header.frame_id = 'camera_depth_frame'
            self.pub_depth.publish(msg)
        
        # 彩色
        color_frame = frames.get_color_frame()
        if color_frame:
            color_img = np.asanyarray(color_frame.get_data())
            msg = self.bridge.cv2_to_imgmsg(color_img, encoding='bgr8')
            msg.header.stamp = self.get_clock().now().to_msg()
            msg.header.frame_id = 'camera_color_frame'
            self.pub_color.publish(msg)
        
        # IMU
        accel_frame = frames.first_or_default(rs.stream.accel)
        gyro_frame = frames.first_or_default(rs.stream.gyro)
        if accel_frame and gyro_frame:
            a = accel_frame.as_motion_frame().get_motion_data()
            g = gyro_frame.as_motion_frame().get_motion_data()
            
            imu_msg = Imu()
            imu_msg.header.stamp = self.get_clock().now().to_msg()
            imu_msg.header.frame_id = 'camera_imu_frame'
            imu_msg.linear_acceleration.x = a.x
            imu_msg.linear_acceleration.y = a.y
            imu_msg.linear_acceleration.z = a.z
            imu_msg.angular_velocity.x = g.x
            imu_msg.angular_velocity.y = g.y
            imu_msg.angular_velocity.z = g.z
            self.pub_imu.publish(imu_msg)


def main(args=None):
    rclpy.init(args=args)
    node = RealSenseBridge()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

## 9. 完整项目模板

```python
"""
完整 SLAM 数据采集 + 实时显示 项目模板
"""
import pyrealsense2 as rs
import numpy as np
import cv2
import json
from datetime import datetime
from pathlib import Path

class RealSenseRecorder:
    def __init__(self, save_dir=None):
        self.pipeline = rs.pipeline()
        self.config = rs.config()
        
        # 启用所有流
        self.config.enable_stream(rs.stream.depth, 640, 480, rs.format.z16, 30)
        self.config.enable_stream(rs.stream.color, 1280, 720, rs.format.bgr8, 30)
        self.config.enable_stream(rs.stream.accel, rs.format.motion_xyz32f, 200)
        self.config.enable_stream(rs.stream.gyro, rs.format.motion_xyz32f, 200)
        
        self.profile = self.pipeline.start(self.config)
        self.device = self.profile.get_device()
        self.serial = self.device.get_info(rs.camera_info.serial_number)
        
        # 对齐与点云
        self.align = rs.align(rs.stream.color)
        self.pc = rs.pointcloud()
        self.colorizer = rs.colorizer()
        
        # 保存路径
        if save_dir is None:
            save_dir = f'capture_{datetime.now():%Y%m%d_%H%M%S}'
        self.save_dir = Path(save_dir)
        self.save_dir.mkdir(exist_ok=True)
        
        # 写 metadata
        metadata = {
            'serial': self.serial,
            'depth_intrinsics': self.profile.get_stream(rs.stream.depth).as_video_stream_profile().get_intrinsics().__dict__,
            'color_intrinsics': self.profile.get_stream(rs.stream.color).as_video_stream_profile().get_intrinsics().__dict__,
        }
        with open(self.save_dir / 'metadata.json', 'w') as f:
            json.dump(metadata, f, indent=2, default=str)
        
        print(f"Recording to {self.save_dir}/")
        print(f"Serial: {self.serial}")
    
    def capture_frame(self):
        frames = self.pipeline.wait_for_frames()
        aligned = self.align.process(frames)
        return aligned
    
    def run(self):
        frame_count = 0
        try:
            while True:
                frames = self.capture_frame()
                
                depth_frame = frames.get_depth_frame()
                color_frame = frames.get_color_frame()
                
                # 显示
                color_img = np.asanyarray(color_frame.get_data())
                depth_colormap = cv2.applyColorMap(
                    cv2.convertScaleAbs(np.asanyarray(depth_frame.get_data()), alpha=0.03),
                    cv2.COLORMAP_JET
                )
                cv2.imshow('Color', color_img)
                cv2.imshow('Depth', depth_colormap)
                
                key = cv2.waitKey(1)
                if key == ord('s'):  # 单帧保存
                    self._save_frame(frames, frame_count)
                    print(f"Saved {frame_count}")
                    frame_count += 1
                elif key & 0xFF == ord('q'):
                    break
        finally:
            self.pipeline.stop()
            cv2.destroyAllWindows()
    
    def _save_frame(self, frames, idx):
        depth_frame = frames.get_depth_frame()
        color_frame = frames.get_color_frame()
        
        cv2.imwrite(str(self.save_dir / f'{idx:06d}_color.png'),
                    np.asanyarray(color_frame.get_data()))
        cv2.imwrite(str(self.save_dir / f'{idx:06d}_depth.png'),
                    np.asanyarray(depth_frame.get_data()))


if __name__ == '__main__':
    recorder = RealSenseRecorder()
    recorder.run()
```
